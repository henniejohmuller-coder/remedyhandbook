# Remedy Handbook — Release Procedure
# Replace X.X.X with new version (e.g. 1.0.10) and N with build number (increment by 1)

## Step 1 — Update version in pubspec.yaml
$content = Get-Content "pubspec.yaml" -Raw
$content = $content -replace "version: 1\.0\.\d+\+\d+", "version: X.X.X+N"
Set-Content "pubspec.yaml" $content

# Example:
$content = Get-Content "pubspec.yaml" -Raw
$content = $content -replace "version: 1\.0\.\d+\+\d+", "version: 1.0.10+12"
Set-Content "pubspec.yaml" $content

## Step 2 — Update version in main.dart
$content = Get-Content "lib\main.dart" -Raw
$content = $content -replace "const currentVersion = '1\.0\.\d+'", "const currentVersion = 'X.X.X'"
Set-Content "lib\main.dart" $content

# Example:
$content = Get-Content "lib\main.dart" -Raw
$content = $content -replace "const currentVersion = '1\.0\.\d+'", "const currentVersion = '1.0.10'"
Set-Content "lib\main.dart" $content

## Step 3 — Update download page APK link
$content = Get-Content "web\download.html" -Raw
$content = $content -replace "V1\.0\.\d+/app-arm64-v8a-release\.apk\?v=[\d\.]+", "VX.X.X/app-arm64-v8a-release.apk?v=X.X.X"
Set-Content "web\download.html" $content

# Example:
$content = Get-Content "web\download.html" -Raw
$content = $content -replace "V1\.0\.\d+/app-arm64-v8a-release\.apk\?v=[\d\.]+", "V1.0.10/app-arm64-v8a-release.apk?v=1.0.10"
Set-Content "web\download.html" $content

## Step 4 — Update Supabase version
# Run in Supabase Dashboard → SQL Editor:
# https://supabase.com/dashboard/project/stawsdjfjzugeleldaxz/sql
UPDATE app_version SET version = 'X.X.X' WHERE id = 1;

# Example:
UPDATE app_version SET version = '1.0.10' WHERE id = 1;

## Step 5 — Deploy web app to Firebase
.\deploy_web.ps1

## Step 6 — Commit and push to GitHub
git add .
git commit -m "Release VX.X.X"
git push origin main

# Example:
git add .
git commit -m "Release V1.0.10"
git push origin main

## Step 7 — Build Android APK
flutter build apk --release --split-per-abi
# Output: android\app\build\outputs\apk\release\app-arm64-v8a-release.apk
# Note: "Gradle build failed" message is cosmetic — APK IS built successfully

## Step 8 — Upload APK to GitHub Releases
